const STATUS_STYLES = {
  verified: { border: "border-emerald-600", text: "text-emerald-700", bg: "bg-emerald-50", label: "Verified" },
  completed: { border: "border-emerald-600", text: "text-emerald-700", bg: "bg-emerald-50", label: "Completed" },
  uploaded: { border: "border-amber-500", text: "text-amber-700", bg: "bg-amber-50", label: "Uploaded" },
  "in progress": { border: "border-amber-500", text: "text-amber-700", bg: "bg-amber-50", label: "In progress" },
  pending: { border: "border-stone-300", text: "text-stone-500", bg: "bg-stone-50", label: "Pending" },
  missing: { border: "border-rose-500", text: "text-rose-700", bg: "bg-rose-50", label: "Missing" },
  delayed: { border: "border-orange-600", text: "text-orange-700", bg: "bg-orange-50", label: "Delayed" },
  rejected: { border: "border-rose-600", text: "text-rose-700", bg: "bg-rose-50", label: "Rejected" },
};

function StatusPill({ status }) {
  const s = STATUS_STYLES[status] || STATUS_STYLES.pending;
  return (
    <span className={`text-xs font-medium px-2 py-0.5 rounded-full ${s.bg} ${s.text}`}>
      {s.label}
    </span>
  );
}

function StatusIcon({ status }) {
  if (status === "verified" || status === "completed")
    return <CheckCircle2 className="w-5 h-5 text-emerald-600 shrink-0" />;
  if (status === "missing" || status === "rejected")
    return <XCircle className="w-5 h-5 text-rose-500 shrink-0" />;
  if (status === "delayed")
    return <AlertTriangle className="w-5 h-5 text-orange-600 shrink-0" />;
  if (status === "uploaded" || status === "in progress")
    return <Clock className="w-5 h-5 text-amber-600 shrink-0" />;
  return <Circle className="w-5 h-5 text-stone-300 shrink-0" />;
}

function ProgressBar({ percent, tone = "slate" }) {
  const bar = tone === "emerald" ? "bg-emerald-600" : tone === "amber" ? "bg-amber-600" : "bg-slate-800";
  return (
    <div className="w-full h-2 rounded-full bg-stone-200 overflow-hidden">
      <div className={`h-full ${bar} rounded-full`} style={{ width: `${percent}%` }} />
    </div>
  );
}

function ScreenHeader({ title, subtitle, onBack }) {
  return (
    <div className="flex items-center gap-3 px-5 pt-6 pb-4 bg-white border-b border-stone-200 sticky top-0 z-10">
      {onBack && (
        <button
          onClick={onBack}
          aria-label="Go back"
          className="w-8 h-8 flex items-center justify-center rounded-full border border-stone-200 text-slate-700"
        >
          <ChevronLeft className="w-4 h-4" />
        </button>
      )}
      <div>
        <h1 className="font-serif text-lg text-slate-900 leading-tight">{title}</h1>
        {subtitle && <p className="text-xs text-stone-500 mt-0.5">{subtitle}</p>}
      </div>
    </div>
  );
}

function Card({ children, className = "" }) {
  return (
    <div className={`bg-white border border-stone-200 rounded-xl p-4 ${className}`}>
      {children}
    </div>
  );
}

function PrimaryButton({ children, onClick, disabled }) {
  return (
    <button
      onClick={onClick}
      disabled={disabled}
      className={`w-full py-3 rounded-lg font-medium text-white text-sm transition-colors ${
        disabled ? "bg-stone-300" : "bg-slate-900 active:bg-slate-800"
      }`}
    >
      {children}
    </button>
  );
}

function SecondaryButton({ children, onClick }) {
  return (
    <button
      onClick={onClick}
      className="w-full py-3 rounded-lg font-medium text-sm border border-slate-900 text-slate-900"

    </button>

}

function LoanPaymentDashboard({ onBack }) {
  const [extra, setExtra] = useState(0);
  const loan = { amount: 7000000, rate: 8.5, tenureYears: 20, emi: 60724, paidMonths: 14, totalMonths: 240 };
  const principalPaid = Math.round((loan.paidMonths / loan.totalMonths) * loan.amount * 0.28);
  const outstanding = loan.amount - principalPaid;

  const simulated = useMemo(() => {
    if (!extra) return null;
    const monthsSaved = Math.round((extra / loan.emi) * (loan.totalMonths - loan.paidMonths) * 0.35);
    const interestSaved = Math.round(monthsSaved * loan.emi * 0.5);
    return { monthsSaved, interestSaved };
  }, [extra]);

  return (
    <div className="pb-24">
      <ScreenHeader title="Loan payments" subtitle="SBI Home Loans · #HP-4471" onBack={onBack} />
      <div className="px-5 pt-4 space-y-4">
        <Card>
          <div className="flex justify-between text-sm mb-1">
            <span className="text-stone-500">EMIs paid</span>
            <span className="text-slate-800">{loan.paidMonths} of {loan.totalMonths}</span>
          </div>
          <ProgressBar percent={(loan.paidMonths / loan.totalMonths) * 100} />
        </Card>

        <Card>
          <div className="grid grid-cols-2 gap-y-3 text-sm">
            <div><p className="text-xs text-stone-400">Loan amount</p><p className="text-slate-700">{money(loan.amount)}</p></div>
            <div><p className="text-xs text-stone-400">Interest rate</p><p className="text-slate-700">{loan.rate}% p.a.</p></div>
            <div><p className="text-xs text-stone-400">Monthly EMI</p><p className="text-slate-700">{money(loan.emi)}</p></div>
            <div><p className="text-xs text-stone-400">Principal paid</p><p className="text-slate-700">{money(principalPaid)}</p></div>
            <div><p className="text-xs text-stone-400">Outstanding</p><p className="text-slate-700">{money(outstanding)}</p></div>
            <div><p className="text-xs text-stone-400">Expected closure</p><p className="text-slate-700">June 2046</p></div>
          </div>
        </Card>

        <Card>
          <h3 className="font-serif text-sm text-slate-900 mb-2.5">Prepayment simulator</h3>
          <label className="text-xs text-stone-500">Extra amount per month</label>
          <input
            type="number"
            value={extra || ""}
            onChange={(e) => setExtra(Number(e.target.value) || 0)}
            placeholder="e.g. 5000"
            className="w-full mt-1 mb-3 px-3 py-2 rounded-lg border border-stone-200 text-sm outline-none focus:border-slate-900"
          />
          {simulated && (
            <div className="bg-emerald-50 rounded-lg p-3 text-sm text-emerald-800 space-y-1">
              <p>Loan closes about {simulated.monthsSaved} months earlier</p>
              <p>Estimated interest saved: {money(simulated.interestSaved)}</p>
            </div>
          )}
          <p className="text-xs text-stone-400 mt-2">Estimates only — actual savings depend on the bank's amortisation schedule.</p>
        </Card>
      </div>
    </div>
  );
}

/* ------------------------------------------------------------------ */
/*  APP ROOT — stack-based navigation                                  */
/* ------------------------------------------------------------------ */

export default function HomePilotApp() {
  const [onboarded, setOnboarded] = useState(false);
  const [stack, setStack] = useState(["home"]);
  const [selectedProperty, setSelectedProperty] = useState(PROPERTIES[2]);

  const current = stack[stack.length - 1];
  const root = stack[0];

  const push = (screen) => setStack((s) => [...s, screen]);
  const pop = () => setStack((s) => (s.length > 1 ? s.slice(0, -1) : s));
  const goTab = (tab) => setStack([tab]);

  if (!onboarded) return <Onboarding onDone={() => setOnboarded(true)} />;

  let content;
  switch (current) {
    case "home":
      content = <HomeDashboard goTab={goTab} push={push} />;
      break;
    case "properties":
      content = <PropertiesList onOpen={(p) => { setSelectedProperty(p); push("propertyDetail"); }} />;
      break;
    case "propertyDetail":
      content = (
        <PropertyDetail
          property={selectedProperty}
          onBack={pop}
          onCheckLoan={() => setStack(["loanMenu", "loanReadiness"])}
        />
      );
      break;
    case "loanMenu":
      content = <LoanMenu push={push} />;
      break;
    case "loanReadiness":
      content = <LoanReadiness onBack={pop} onCompareBanks={() => push("bankCompare")} />;
      break;
    case "bankCompare":
      content = <BankCompare onBack={pop} onChoose={() => push("checklist")} />;
      break;
    case "checklist":
      content = <DocumentChecklist onBack={pop} />;
      break;
    case "tracker":
      content = <LoanTracker onBack={pop} />;
      break;
    case "officerVisit":
      content = <OfficerVisit onBack={pop} />;
      break;
    case "payment":
      content = <LoanPaymentDashboard onBack={pop} />;
      break;
    case "documents":
      content = <DocumentVault />;
      break;
    case "profile":
      content = <Profile />;
      break;
    default:
      content = <HomeDashboard goTab={goTab} push={push} />;
  }

  return (
    <div className="max-w-md mx-auto min-h-screen bg-stone-50 font-sans relative">
      {content}
      <BottomNav activeRoot={root} onSelect={goTab} />
    </div>
  );
}
